One-time Supabase setup

1Create a free account at supabase.com and start a new project.
2In your project, open SQL Editor and run this query to create tables and row-level security:
-- Sessions table CREATE EXTENSION IF NOT EXISTS "uuid-ossp"; CREATE TABLE IF NOT EXISTS sessions ( id UUID PRIMARY KEY DEFAULT uuid_generate_v4(), user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE, session_type TEXT NOT NULL DEFAULT 'practice', session_name TEXT NOT NULL, players JSONB NOT NULL DEFAULT '[]', scoring_rules TEXT NOT NULL DEFAULT 'cancellation', status TEXT NOT NULL DEFAULT 'active', created_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); CREATE TABLE IF NOT EXISTS throws ( id UUID PRIMARY KEY DEFAULT uuid_generate_v4(), session_id UUID NOT NULL REFERENCES sessions(id) ON DELETE CASCADE, user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE, end_number INT NOT NULL, position INT NOT NULL, player_name TEXT NOT NULL, player_team TEXT NOT NULL, shot_type TEXT NOT NULL, thrown_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); -- Indexes CREATE INDEX IF NOT EXISTS idx_sessions_user ON sessions(user_id); CREATE INDEX IF NOT EXISTS idx_throws_session ON throws(session_id); CREATE INDEX IF NOT EXISTS idx_throws_user ON throws(user_id); -- Row Level Security ALTER TABLE sessions ENABLE ROW LEVEL SECURITY; ALTER TABLE throws ENABLE ROW LEVEL SECURITY; CREATE POLICY "own_sessions_select" ON sessions FOR SELECT USING (auth.uid() = user_id); CREATE POLICY "own_sessions_insert" ON sessions FOR INSERT WITH CHECK (auth.uid() = user_id); CREATE POLICY "own_sessions_update" ON sessions FOR UPDATE USING (auth.uid() = user_id); CREATE POLICY "own_sessions_delete" ON sessions FOR DELETE USING (auth.uid() = user_id); CREATE POLICY "own_throws_select" ON throws FOR SELECT USING (auth.uid() = user_id); CREATE POLICY "own_throws_insert" ON throws FOR INSERT WITH CHECK (auth.uid() = user_id); CREATE POLICY "own_throws_update" ON throws FOR UPDATE USING (auth.uid() = user_id); CREATE POLICY "own_throws_delete" ON throws FOR DELETE USING (auth.uid() = user_id);
3Go to Project Settings → API. Copy your Project URL and anon public key below.

The Sport
About Horseshoe Pitching

Horseshoe pitching is one of America's oldest backyard sports — a game of skill, precision, and consistency. Players alternate pitching horseshoes at a stake set 40 feet away (27 feet for women and juniors), trying to ring the stake or land as close as possible.

Shot Values & Scoring

  • Ringer (3 pts) — The shoe encircles the stake. A straight line touching both ends must clear the stake.
  • Leaner (2 pts) — The shoe leans against the stake. Counts as a scoring shoe.
  • Scorer (1 pt) — Shoe lands within 6 inches of the stake but doesn't lean.
  • No Score (0 pts) — All other shoes not meeting scoring criteria.
  • Cancellation Scoring — Only the player closest to the pin scores; equal values cancel out.
  • All-Counting (Count-All) — Both players score the full value of all scoring shoes each end.
  • Standard game plays to 40 points or a set number of ends.
